# Per-TU symbol ownership for GP-relative addressing.
#
# Being inside the +/-32KB $gp window makes a symbol GP-addressable, but ASPSX
# only uses GP-relative addressing for symbols the translation unit declares
# itself; every other TU addresses the same symbol absolutely. The original
# binary does exactly that for the 0x8005E3A4..0x8005E3C0 cluster, which
# func_80011370's TU owns (it stores them with `sw %gp_rel(...)($gp)`) while
# other TUs load them with the absolute self-clobber macro
# (`lui $v0,%hi(SYM)` / `lw $v0,%lo(SYM)($v0)`).
#
# Our headers give every file the same declaration, so ownership cannot be
# expressed in C. List it here instead:
#
#   <source stem> = <symbols this TU does NOT own>
#
# tools/build/fixSmallDataExterns.ts widens those symbols' `.extern` size past
# the -G threshold so GNU as expands the macro absolutely. Symbols outside the
# $gp window need no entry; the address test already covers them.

# Evidence: SetGfxClip/SetGfxOffset targets read D_8005E3AC and D_8005E3A8 as
# `lui $v0,0x8006 / lw $v0,-7252($v0)` and `lui $v1,0x8006 / lw $v1,-7256($v1)`
# — absolute, one register each. func_80011370 writes the same two symbols
# GP-relative, so it is the owning TU.
SetGfxClip = D_8005E3A8 D_8005E3AC
SetGfxOffset = D_8005E3A8 D_8005E3AC

# Evidence: the sprite-renderer TU reads D_8005E3C0 with the absolute
# self-clobber pair (lui $v1,%hi / lw $v1,%lo($v1)) while func_80011370 stores
# it GP-relative (sw $s2,%gp_rel(D_8005E3C0)($gp)) — so func_80011370 owns it
# and these two do not. Before this entry they came out 4 bytes short each,
# because the scalar declaration let GNU as resolve the load GP-relatively.
func_800165D8 = D_8005E3C0
func_80016C08 = D_8005E3C0
